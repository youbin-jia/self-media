# Phase 1 MVP核心功能 实施计划

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现从选题到视频输出的完整最小可行产品，能在30分钟内生成一条5分钟的测试视频

**Architecture:** 采用前后端分离架构，后端FastAPI提供RESTful API，前端React提供Web界面。使用Celery处理异步任务（视频合成、素材采集）。数据库使用SQLite存储项目数据、脚本、素材元数据。模块化设计，每个功能独立成模块。

**Tech Stack:** FastAPI, React, SQLite, Celery, Redis, FFmpeg, MoviePy, Claude API, Pexels API

    async def fetch_topics(self, source: str = None, limit: int = 20) -> List[Dict[str, Any]]:
        """
        获取热点列表
        Args:
            source: 数据源（weibo, zhihu, baidu等），None表示全部
            limit: 返回数量限制
        Returns:
            热点列表
        """
        topics = self.mock_topics

        if source:
            topics = [t for t in topics if t["source"] == source]

        # 按热度排序
        topics = sorted(topics, key=lambda x: x["hot_score"], reverse=True)

        return topics[:limit]

    async def refresh_topics(self) -> Dict[str, Any]:
        """
        刷新热点数据
        Returns:
            刷新结果
        """
        # Phase 1: 模拟刷新，随机调整热度
        for topic in self.mock_topics:
            topic["hot_score"] = min(100, topic["hot_score"] + random.randint(-5, 5))
            topic["created_at"] = datetime.now().isoformat()

        return {
            "status": "success",

# backend/app/services/llm_provider.py
from abc import ABC, abstractmethod
from typing import Optional
import anthropic
from app.config import settings


class LLMProvider(ABC):
    """LLM提供商抽象基类"""

    @abstractmethod
    async def generate(self, prompt: str, **kwargs) -> str:
        """生成文本"""
        pass


class ClaudeProvider(LLMProvider):
    """Claude API提供商"""

    def __init__(self, api_key: Optional[str] = None):
        self.api_key = api_key or settings.ANTHROPIC_API_KEY
        if not self.api_key:
            raise ValueError("ANTHROPIC_API_KEY not configured")
        self.client = anthropic.AsyncAnthropic(api_key=self.api_key)

    async def generate(
        self,
        prompt: str,
        model: str = "claude-sonnet-4-6-20250514",
        max_tokens: int = 4096,
        temperature: float = 0.7,
        **kwargs
    ) -> str:
        """
        生成文本
        Args:
            prompt: 提示词
            model: 模型名称
            max_tokens: 最大token数
            temperature: 温度参数
        Returns:
            生成的文本
        """
        try:
            message = await self.client.messages.create(
                model=model,
                max_tokens=max_tokens,
                temperature=temperature,
                messages=[{"role": "user", "content": prompt}]
            )
            return message.content[0].text
        except Exception as e:
            raise RuntimeError(f"Claude API error: {str(e)}")


# LLM Provider工厂
def get_llm_provider(provider: str = "claude") -> LLMProvider:
    """
    获取LLM Provider实例
    Args:
        provider: 提供商名称
    Returns:
        LLMProvider实例
    """
    if provider == "claude":
        return ClaudeProvider()
    else:
        raise ValueError(f"Unsupported LLM provider: {provider}")

# backend/app/services/quality_detector.py
from typing import Dict, Any, List
from decimal import Decimal


class QualityDetector:
    """质量检测服务"""

    async def detect_script_quality(
        self,
        script_text: str,
        segments: List[Dict[str, Any]]
    ) -> Dict[str, Any]:
        """
        检测脚本质量
        Args:
            script_text: 完整脚本文本
            segments: 脚本分段
        Returns:
            质量报告
        """
        # 计算指标
        word_count = len(script_text)
        segment_count = len(segments)
        avg_duration = sum(s["duration"] for s in segments) / max(segment_count, 1)

        # 简单评分逻辑
        score = 0.0
        metrics = {}
        issues = []

        # 信息密度评分 (30分)
        density_score = min(30, word_count / 50)  # 每50字1分，最高30分
        metrics["word_count"] = word_count
        metrics["density_score"] = density_score
        score += density_score

        # 时长合理性评分 (20分)
        if 3 <= avg_duration <= 10:
            duration_score = 20
        elif 2 <= avg_duration <= 15:
            duration_score = 15
        else:
            duration_score = 10
            issues.append({
                "type": "duration",
                "message": f"平均段落时长{avg_duration:.1f}秒不在理想范围(3-10秒)"
            })
        metrics["avg_duration"] = avg_duration
        metrics["duration_score"] = duration_score
        score += duration_score

        # 情绪多样性评分 (20分)
        emotions = set(s.get("emotion") for s in segments if s.get("emotion"))
        emotion_score = min(20, len(emotions) * 5)
        metrics["emotion_count"] = len(emotions)
        metrics["emotion_score"] = emotion_score
        score += emotion_score

        # 总分
        overall_score = Decimal(str(score)).quantize(Decimal("0.01"))
        grade = self._calculate_grade(overall_score)

        return {
            "overall_score": overall_score,
            "grade": grade,
            "metrics": metrics,
            "issues": issues,
            "recommendations": self._generate_recommendations(grade, issues)
        }

    def _calculate_grade(self, score: Decimal) -> str:
        """计算等级"""
        if score >= 90:
            return "A"
        elif score >= 80:
            return "B"
        elif score >= 70:
            return "C"
        elif score >= 60:
            return "D"
        else:
            return "E"

    def _generate_recommendations(self, grade: str, issues: List[Dict]) -> List[str]:
        """生成改进建议"""
        recommendations = []

        if grade in ["D", "E"]:
            recommendations.append("建议重新生成脚本，调整prompt参数")

        for issue in issues:
            if issue["type"] == "duration":
                recommendations.append("调整脚本分段，使每个段落时长更合理")

        if grade == "B":
            recommendations.append("脚本质量良好，可直接使用或小幅优化")

        return recommendations

# backend/app/services/video_synthesizer.py
from typing import List, Dict, Any
from pathlib import Path
from moviepy.editor import ImageClip, concatenate_videoclips, AudioFileClip
import uuid
from app.config import settings


class VideoSynthesizer:
    """视频合成服务 - Phase 1简单版本"""

    def __init__(self):
        self.output_dir = Path(settings.DATA_DIR) / "videos"
        self.output_dir.mkdir(parents=True, exist_ok=True)

    async def synthesize(
        self,
        project_id: str,
        segments: List[Dict[str, Any]],
        materials: List[Dict[str, Any]]
    ) -> Dict[str, Any]:
        """
        合成视频
        Args:
            project_id: 项目ID
            segments: 脚本分段
            materials: 素材列表
        Returns:
            合成结果
        """
        # Phase 1: 简单图片拼接
        clips = []

        for segment in segments:
            # 为每个段落分配一个素材
            material = materials[len(clips) % len(materials)] if materials else None

            if material and material.get("local_path"):
                # 使用实际素材
                clip = ImageClip(material["local_path"], duration=segment["duration"])
            else:
                # 使用占位符（纯色背景）
                clip = ImageClip("placeholder.jpg", duration=segment["duration"])

            clips.append(clip)

        # 拼接所有片段
        final_video = concatenate_videoclips(clips, method="compose")

        # 输出视频
        output_filename = f"{project_id}_preview.mp4"
        output_path = self.output_dir / output_filename

        final_video.write_videofile(
            str(output_path),
            fps=24,
            codec="libx264",
            audio=False  # Phase 1暂不处理音频
        )

        return {
            "video_path": str(output_path),
            "duration": final_video.duration,
            "format": "mp4",
            "resolution": "1920x1080"
        }

@celery_app.task(bind=True)
def synthesize_video_task(self, project_id: str):
    """视频合成任务"""
    db = SessionLocal()

    try:
        # 更新任务进度
        task = Task(
            project_id=project_id,
            task_type="video_synthesis",
            status="running",
            progress=0
        )
        db.add(task)
        db.commit()

        # 获取项目数据
        project = db.query(Project).filter(Project.id == project_id).first()
        script = db.query(Script).filter(
            Script.project_id == project_id,
            Script.is_approved == True
        ).first()
        materials = db.query(Material).filter(Material.project_id == project_id).all()

        # 更新进度: 准备数据 (20%)
        task.progress = 20
        db.commit()

        # 合成视频
        result = video_synthesizer.synthesize(
            project_id=project_id,
            segments=script.segments,
            materials=[{"local_path": m.local_path} for m in materials]
        )

        # 更新进度: 合成完成 (80%)
        task.progress = 80
        db.commit()

        # 更新项目状态
        project.status = "preview_ready"
        project.current_step = "video_preview"
        project.metadata["video_path"] = result["video_path"]

        # 更新任务状态
        task.status = "completed"
        task.progress = 100
        task.result = result
        db.commit()

        return {"status": "success", "video_path": result["video_path"]}

    except Exception as e:
        task.status = "failed"
        task.error_message = str(e)
        db.commit()
        raise

function ProjectWorkflow() {
